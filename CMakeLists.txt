cmake_minimum_required(VERSION 3.28)
set(CMAKE_OSX_DEPLOYMENT_TARGET "13.3" CACHE STRING "Minimum macOS")
project(PrisonLife LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LAM_BUILD_TESTS ON CACHE BOOL "Engine tests")
set(LAM_BUILD_EXAMPLES ON CACHE BOOL "Engine examples")
set(LAM_LTO OFF CACHE BOOL "Fast iteration")
add_subdirectory(LambdaEngine2D)
lam_add_game(NAME prison_life SOURCES src/main.cpp src/view.cpp ASSETS assets
  SIM_LIBRARY prison_sim
  SIM_SOURCES src/model.cpp src/map.cpp src/game.cpp src/save.cpp src/interactions.cpp src/people.cpp src/broadcast.cpp
  SIM_DEPS lam_app lam_ecs lam_input lam_physics lam_tilemap lam_scene lam_ui lam_audio)
target_include_directories(prison_life PRIVATE src)
lam_add_game(NAME prison_verify SOURCES src/main.cpp src/view.cpp tools/verify.cpp DEPS lam_engine prison_sim)
target_include_directories(prison_verify PRIVATE src)
target_compile_definitions(prison_verify PRIVATE PRISON_VERIFICATION=1)
add_dependencies(prison_verify prison_life_assets)
enable_testing()
add_executable(prison_tests tests/test_rules.cpp src/view.cpp)
target_link_libraries(prison_tests PRIVATE prison_sim lam_engine doctest::doctest)
target_include_directories(prison_tests PRIVATE src)
add_test(NAME prison_rules COMMAND prison_tests)
set_tests_properties(prison_rules PROPERTIES LABELS prison)
